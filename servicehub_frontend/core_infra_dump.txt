import 'dart:async';
import 'dart:convert';
import 'package:http/http.dart' as http;

import '../constants/api_constants.dart';
import 'api_exception.dart';
import 'token_storage.dart';

class ApiService {
  final TokenStorage _tokenStorage;
  final http.Client _client;
  final String _baseUrl;

  ApiService({
    TokenStorage? tokenStorage,
    http.Client? client,
    String? baseUrl,
  })  : _tokenStorage = tokenStorage ?? const TokenStorage(),
        _client = client ?? http.Client(),
        _baseUrl = baseUrl ?? ApiConstants.baseUrl;

  static const Duration _timeoutDuration = Duration(seconds: 10);

  Future<Map<String, String>> _buildHeaders({
    bool authenticated = false,
    Map<String, String>? additionalHeaders,
  }) async {
    final headers = <String, String>{
      'Accept': 'application/json',
      'Content-Type': 'application/json',
      ...?additionalHeaders,
    };

    if (authenticated) {
      final token = await _tokenStorage.getToken();
      if (token == null || token.isEmpty) {
        throw const ApiException(
          message: 'Authentication token not found.',
          statusCode: 401,
        );
      }
      headers['Authorization'] = 'Bearer $token';
    }

    return headers;
  }

  Uri _buildUri(String endpoint) {
    final cleanEndpoint = endpoint.startsWith('/') ? endpoint : '/$endpoint';
    return Uri.parse('$_baseUrl$cleanEndpoint');
  }

  Future<http.Response> _sendRequest(Future<http.Response> Function() request) async {
    try {
      final response = await request().timeout(_timeoutDuration);
      return _processResponse(response);
    } on TimeoutException {
      throw const ApiException(
        message: 'Request timed out. Please check your connection and try again.',
        statusCode: 408,
      );
    } on http.ClientException catch (e) {
      throw ApiException(
        message: 'Network error: ${e.message}',
      );
    } on ApiException {
      rethrow;
    } catch (e) {
      throw ApiException(
        message: 'Unexpected network error: $e',
      );
    }
  }

  http.Response _processResponse(http.Response response) {
    if (response.statusCode >= 200 && response.statusCode < 300) {
      return response;
    }

    String message = 'Request failed with status ${response.statusCode}';
    Map<String, List<String>>? validationErrors;
    dynamic bodyData;

    try {
      if (response.body.isNotEmpty) {
        bodyData = jsonDecode(response.body);
        if (bodyData is Map<String, dynamic>) {
          if (bodyData['message'] is String && (bodyData['message'] as String).isNotEmpty) {
            message = bodyData['message'] as String;
          } else if (bodyData['error'] is Map && bodyData['error']['message'] is String) {
            message = bodyData['error']['message'] as String;
          } else if (bodyData['error'] is String) {
            message = bodyData['error'] as String;
          }

          // Parse Laravel validation error schema (422)
          final rawErrors = bodyData['errors'] ??
              (bodyData['error'] is Map ? bodyData['error']['details'] : null);
          if (rawErrors is Map<String, dynamic>) {
            validationErrors = {};
            for (final entry in rawErrors.entries) {
              if (entry.value is List) {
                validationErrors[entry.key] =
                    (entry.value as List).map((e) => e.toString()).toList();
              } else if (entry.value is String) {
                validationErrors[entry.key] = [entry.value as String];
              }
            }
          }
        }
      }
    } catch (_) {
      // Body is not JSON (e.g., HTML 500 error)
    }

    throw ApiException(
      message: message,
      statusCode: response.statusCode,
      body: bodyData ?? response.body,
      errors: validationErrors,
    );
  }

  Future<http.Response> get(
    String endpoint, {
    bool authenticated = false,
    Map<String, String>? headers,
  }) async {
    final finalHeaders = await _buildHeaders(
      authenticated: authenticated,
      additionalHeaders: headers,
    );
    return _sendRequest(() => _client.get(_buildUri(endpoint), headers: finalHeaders));
  }

  Future<http.Response> post(
    String endpoint, {
    dynamic body,
    bool authenticated = false,
    Map<String, String>? headers,
  }) async {
    final finalHeaders = await _buildHeaders(
      authenticated: authenticated,
      additionalHeaders: headers,
    );
    final encodedBody = body is String ? body : (body != null ? jsonEncode(body) : null);
    return _sendRequest(
      () => _client.post(_buildUri(endpoint), headers: finalHeaders, body: encodedBody),
    );
  }

  Future<http.Response> put(
    String endpoint, {
    dynamic body,
    bool authenticated = false,
    Map<String, String>? headers,
  }) async {
    final finalHeaders = await _buildHeaders(
      authenticated: authenticated,
      additionalHeaders: headers,
    );
    final encodedBody = body is String ? body : (body != null ? jsonEncode(body) : null);
    return _sendRequest(
      () => _client.put(_buildUri(endpoint), headers: finalHeaders, body: encodedBody),
    );
  }

  Future<http.Response> patch(
    String endpoint, {
    dynamic body,
    bool authenticated = false,
    Map<String, String>? headers,
  }) async {
    final finalHeaders = await _buildHeaders(
      authenticated: authenticated,
      additionalHeaders: headers,
    );
    final encodedBody = body is String ? body : (body != null ? jsonEncode(body) : null);
    return _sendRequest(
      () => _client.patch(_buildUri(endpoint), headers: finalHeaders, body: encodedBody),
    );
  }

  Future<http.Response> delete(
    String endpoint, {
    dynamic body,
    bool authenticated = false,
    Map<String, String>? headers,
  }) async {
    final finalHeaders = await _buildHeaders(
      authenticated: authenticated,
      additionalHeaders: headers,
    );
    final encodedBody = body is String ? body : (body != null ? jsonEncode(body) : null);
    return _sendRequest(
      () => _client.delete(_buildUri(endpoint), headers: finalHeaders, body: encodedBody),
    );
  }

  // Convenience aliases for authenticated calls
  Future<http.Response> authenticatedGet(
    String endpoint, {
    Map<String, String>? headers,
  }) =>
      get(endpoint, authenticated: true, headers: headers);

  Future<http.Response> authenticatedPost(
    String endpoint, {
    dynamic body,
    Map<String, String>? headers,
  }) =>
      post(endpoint, body: body, authenticated: true, headers: headers);

  Future<http.Response> authenticatedPatch(
    String endpoint, {
    dynamic body,
    Map<String, String>? headers,
  }) =>
      patch(endpoint, body: body, authenticated: true, headers: headers);

  Future<http.Response> authenticatedDelete(
    String endpoint, {
    dynamic body,
    Map<String, String>? headers,
  }) =>
      delete(endpoint, body: body, authenticated: true, headers: headers);
}

class ApiException implements Exception {
  final String message;
  final int? statusCode;
  final dynamic body;
  final Map<String, List<String>>? errors;

  const ApiException({
    required this.message,
    this.statusCode,
    this.body,
    this.errors,
  });

  /// Extracts the first error message or returns the general message.
  String get firstErrorMessage {
    if (errors != null && errors!.isNotEmpty) {
      final firstKey = errors!.keys.first;
      final firstList = errors![firstKey];
      if (firstList != null && firstList.isNotEmpty) {
        return firstList.first;
      }
    }
    return message;
  }

  @override
  String toString() {
    if (statusCode != null) {
      return 'ApiException ($statusCode): $message';
    }
    return 'ApiException: $message';
  }
}

import 'package:flutter_secure_storage/flutter_secure_storage.dart';

class TokenStorage {
  final FlutterSecureStorage _storage;

  const TokenStorage({FlutterSecureStorage? storage})
      : _storage = storage ?? const FlutterSecureStorage();

  static const String _tokenKey = 'auth_token';

  Future<void> saveToken(String token) async {
    await _storage.write(key: _tokenKey, value: token);
  }

  Future<String?> getToken() async {
    return await _storage.read(key: _tokenKey);
  }

  Future<void> deleteToken() async {
    await _storage.delete(key: _tokenKey);
  }

  Future<bool> hasToken() async {
    final token = await getToken();
    return token != null && token.isNotEmpty;
  }
}

class AuthRedirect {
  const AuthRedirect._();

  /// Routes that guests (unauthenticated users) may access freely.
  static const Set<String> _guestRoutes = {
    '/landing',
    '/login',
    '/register',
    '/service-categories',
  };

  static String? redirect({
    required bool isAuthenticated,
    bool isLoading = false,
    required String location,
    String? fullLocation,
    String? redirectParam,
  }) {
    // Wait for session restoration on splash
    if (isLoading && location == '/') {
      return null;
    }

    final isGuestRoute = _guestRoutes.contains(location);

    if (!isAuthenticated) {
      // After the splash check completes, unauthenticated users go to landing
      if (location == '/') {
        return '/landing';
      }
      // Guest routes are allowed
      if (isGuestRoute) {
        return null;
      }
      // Everything else requires auth â€” preserve destination for post-login
      final intended = fullLocation ?? location;
      final encoded = Uri.encodeComponent(intended);
      return '/login?redirect=$encoded';
    }

    // Authenticated users should not see splash, landing, or auth screens
    if (location == '/login' || location == '/register') {
      // GoRouter already decodes query params â€” use redirectParam directly
      if (redirectParam != null && redirectParam.isNotEmpty) {
        return redirectParam;
      }
      return '/customer-home';
    }

    if (location == '/' || location == '/landing') {
      return '/customer-home';
    }

    return null;
  }
}

import 'package:flutter_riverpod/flutter_riverpod.dart';

import 'core/localization/locale_provider.dart';
import 'features/auth/presentation/providers/auth_providers.dart';
import 'l10n/app_localizations.dart';
import 'ui/core/router/app_router.dart';
import 'ui/core/theme/app_theme.dart';

void main() {
  WidgetsFlutterBinding.ensureInitialized();
  runApp(
    const ProviderScope(
      child: ServiceHubApp(),
    ),
  );
}

class ServiceHubApp extends ConsumerStatefulWidget {
  const ServiceHubApp({super.key});

  @override
  ConsumerState<ServiceHubApp> createState() => _ServiceHubAppState();
}

class _ServiceHubAppState extends ConsumerState<ServiceHubApp> {
  @override
  void initState() {
    super.initState();
    Future.microtask(() {
      ref.read(authStateProvider.notifier).checkAuth();
    });
  }

  @override
  Widget build(BuildContext context) {
    final router = ref.watch(appRouterProvider);
    final locale = ref.watch(localeProvider);

    return MaterialApp.router(
      debugShowCheckedModeBanner: false,
      title: 'ServiceHUB',
      theme: AppTheme.light(),
      darkTheme: AppTheme.dark(),
      themeMode: ThemeMode.light,
      locale: locale,
      supportedLocales: LocaleNotifier.supportedLocales,
      localizationsDelegates: const [
        AppLocalizations.delegate,
        GlobalMaterialLocalizations.delegate,
        GlobalWidgetsLocalizations.delegate,
        GlobalCupertinoLocalizations.delegate,
      ],
      routerConfig: router,
    );
  }
}
