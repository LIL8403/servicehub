class ArtisanSelectionRepository {
  ArtisanSelectionRepository({required this.apiService});

  final ApiService apiService;

  Future<EligibleArtisansResponse> fetchEligibleArtisans(
    String requestId,
  ) async {
    try {
      final response = await apiService.authenticatedGet(
        '/customer/service-requests/$requestId/eligible-artisans',
      );

      final json = jsonDecode(response.body) as Map<String, dynamic>;
      return EligibleArtisansResponse.fromJson(json);
    } on ApiException {
      rethrow;
    } catch (e) {
      throw ApiException(message: 'Failed to load artisans: $e');
    }
  }

  Future<Map<String, dynamic>> selectArtisan({
    required String requestId,
    required String artisanId,
  }) async {
    try {
      final response = await apiService.authenticatedPost(
        '/customer/service-requests/$requestId/select-artisan',
        body: {'artisan_profile_id': artisanId},
      );

      final json = jsonDecode(response.body) as Map<String, dynamic>;
      return json;
    } on ApiException {
      rethrow;
    } catch (e) {
      throw ApiException(message: 'Failed to select artisan: $e');
    }
  }
}

class ServiceRequestRepository {
  ServiceRequestRepository({required this.apiService});

  final ApiService apiService;

  Future<SubmittedRequest> submit({
    required ServiceRequestDraft draft,
    required String serviceId,
  }) async {
    final body = <String, dynamic>{
      'service_id': serviceId,
      'description': draft.description.trim(),
      'ai_assist_enabled': draft.aiAssistEnabled,
      if (draft.urgency != null) 'urgency': _urgencyApiValue(draft.urgency!),
      'location': {
        'label': draft.formattedAddress?.trim() ?? '',
        'type': 'Point',
        'coordinates': [
          draft.locationLongitude ?? 0.0,
          draft.locationLatitude ?? 0.0,
        ],
      },
      if (draft.preferredDate != null || draft.isFlexible)
        'preferred_schedule': {
          'date': draft.preferredDate != null
              ? _formatDate(draft.preferredDate!)
              : null,
          'window': _scheduleWindow(draft),
          'notes': draft.additionalNotes.trim().isEmpty
              ? null
              : draft.additionalNotes.trim(),
        },
    };

    // Add this just before the try block
    // ignore: avoid_print
    print('SUBMIT PAYLOAD: ${jsonEncode(body)}');

    try {
      final response = await apiService.authenticatedPost(
        '/customer/service-requests',
        body: body,
      );

      final json = jsonDecode(response.body) as Map<String, dynamic>;
      final request = json['service_request'] as Map<String, dynamic>;

      return SubmittedRequest(
        id: request['_id']?.toString() ?? request['id']?.toString() ?? '',
        status: request['status']?.toString() ?? 'unknown',
        aiAssistEnabled: request['ai_assist_enabled'] == true,
      );
    } on ApiException {
      rethrow;
    } catch (e) {
      throw ApiException(message: 'Failed to submit request: $e');
    }
  }

  String _urgencyApiValue(UrgencyLevel urgency) {
    switch (urgency) {
      case UrgencyLevel.low:
        return 'low';
      case UrgencyLevel.medium:
        return 'medium';
      case UrgencyLevel.high:
        return 'high';
    }
  }

  String? _scheduleWindow(ServiceRequestDraft draft) {
    if (draft.isFlexible) {
      return 'Flexible';
    }
    final earliest = draft.preferredStartTime;
    final latest = draft.preferredEndTime;
    if (earliest != null && latest != null) {
      return '$earliest â€“ $latest';
    }
    if (earliest != null) {
      return 'After $earliest';
    }
    if (latest != null) {
      return 'Before $latest';
    }
    return null;
  }

  String _formatDate(DateTime date) {
    final y = date.year.toString().padLeft(4, '0');
    final m = date.month.toString().padLeft(2, '0');
    final d = date.day.toString().padLeft(2, '0');
    return '$y-$m-$d';
  }
}

class EligibleArtisansResponse {
  const EligibleArtisansResponse({
    required this.artisans,
    required this.aiRecommendations,
    required this.request,
  });

  final List<ArtisanSummary> artisans;
  final List<AiRecommendation> aiRecommendations;
  final RequestSnapshot request;

  /// True when the backend has stored AI rankings for this request.
  bool get hasAiRecommendations =>
      request.aiRecommendationsAvailable && aiRecommendations.isNotEmpty;

  /// Returns the artisan matching the given id, or null.
  ArtisanSummary? artisanById(String id) {
    for (final a in artisans) {
      if (a.id == id) return a;
    }
    return null;
  }

  /// The AI-ranked artisans, in order of match score descending,
  /// resolved to their full ArtisanSummary objects. Any ranking whose
  /// artisan_id does not match an artisan in the list is skipped.
  List<({ArtisanSummary artisan, AiRecommendation recommendation})>
      get rankedArtisans {
    final result =
        <({ArtisanSummary artisan, AiRecommendation recommendation})>[];
    for (final rec in aiRecommendations) {
      final a = artisanById(rec.artisanId);
      if (a != null) {
        result.add((artisan: a, recommendation: rec));
      }
    }
    return result;
  }

  /// The artisans not present in the AI recommendations list.
  List<ArtisanSummary> get otherArtisans {
    final rankedIds = aiRecommendations.map((r) => r.artisanId).toSet();
    return artisans.where((a) => !rankedIds.contains(a.id)).toList();
  }

  factory EligibleArtisansResponse.fromJson(Map<String, dynamic> json) {
    final list = (json['artisans'] as List?) ?? [];
    final aiList = (json['ai_recommendations'] as List?) ?? [];
    final requestJson = (json['request'] as Map<String, dynamic>?) ?? {};

    return EligibleArtisansResponse(
      artisans: list
          .whereType<Map<String, dynamic>>()
          .map((a) => ArtisanSummary.fromJson(a))
          .toList(),
      aiRecommendations: aiList
          .whereType<Map<String, dynamic>>()
          .map((r) => AiRecommendation.fromJson(r))
          .toList(),
      request: RequestSnapshot.fromJson(requestJson),
    );
  }
}

/// Holds the mutable state of a service request as it is being built
/// through the multi-step wizard. This is a plain data class; the wizard's
/// provider copies it via [copyWith] on every edit.
@immutable
class ServiceRequestDraft {
  const ServiceRequestDraft({
    this.categoryId,
    this.description = '',
    this.photoFiles = const [],
    this.urgency,
    this.aiAssistEnabled = false,
    this.locationLatitude,
    this.locationLongitude,
    this.formattedAddress,
    this.preferredDate,
    this.preferredStartTime,
    this.preferredEndTime,
    this.isFlexible = false,
    this.additionalNotes = '',
  });

  final String? categoryId;
  final String description;
  final List<File> photoFiles;
  final UrgencyLevel? urgency;
  final bool aiAssistEnabled;

  final double? locationLatitude;
  final double? locationLongitude;
  final String? formattedAddress;

  final DateTime? preferredDate;
  final String? preferredStartTime;
  final String? preferredEndTime;
  final bool isFlexible;
  final String additionalNotes;

  /// True when Step 1 is valid enough to continue.
  bool get isStep1Valid =>
      categoryId != null &&
      description.trim().isNotEmpty &&
      urgency != null;

  /// True when Step 2 is valid enough to continue.
  bool get isStep2Valid =>
      locationLatitude != null &&
      locationLongitude != null &&
      formattedAddress != null;

  /// True when Step 3 is valid enough to continue.
  bool get isStep3Valid {
    if (isFlexible) return true;
    return preferredDate != null &&
        preferredStartTime != null &&
        preferredEndTime != null;
  }

  ServiceRequestDraft copyWith({
    String? categoryId,
    String? description,
    List<File>? photoFiles,
    UrgencyLevel? urgency,
    bool? aiAssistEnabled,
    double? locationLatitude,
    double? locationLongitude,
    String? formattedAddress,
    DateTime? preferredDate,
    String? preferredStartTime,
    String? preferredEndTime,
    bool? isFlexible,
    String? additionalNotes,
  }) {
    return ServiceRequestDraft(
      categoryId: categoryId ?? this.categoryId,
      description: description ?? this.description,
      photoFiles: photoFiles ?? this.photoFiles,
      urgency: urgency ?? this.urgency,
      aiAssistEnabled: aiAssistEnabled ?? this.aiAssistEnabled,
      locationLatitude: locationLatitude ?? this.locationLatitude,
      locationLongitude: locationLongitude ?? this.locationLongitude,
      formattedAddress: formattedAddress ?? this.formattedAddress,
      preferredDate: preferredDate ?? this.preferredDate,
      preferredStartTime: preferredStartTime ?? this.preferredStartTime,
      preferredEndTime: preferredEndTime ?? this.preferredEndTime,
      isFlexible: isFlexible ?? this.isFlexible,
      additionalNotes: additionalNotes ?? this.additionalNotes,
    );
  }
}

class _ArtisanSelectionPageState extends ConsumerState<ArtisanSelectionPage> {
  _SortOption _sort = _SortOption.topRated;
  bool _verifiedOnly = false;

  List<ArtisanSummary> _applySort(List<ArtisanSummary> input) {
    final list = List<ArtisanSummary>.from(input);
    switch (_sort) {
      case _SortOption.topRated:
        list.sort((a, b) => b.ratingAverage.compareTo(a.ratingAverage));
      case _SortOption.nearest:
        list.sort((a, b) =>
            (a.distanceKm ?? 9999).compareTo(b.distanceKm ?? 9999));
      case _SortOption.mostExperienced:
        list.sort((a, b) => b.experienceYears.compareTo(a.experienceYears));
    }
    return list;
  }

  @override
  Widget build(BuildContext context) {
    final l10n = AppLocalizations.of(context);
    final async = ref.watch(eligibleArtisansProvider(widget.requestId));

    return Scaffold(
      backgroundColor: AppColors.bgLight,
      appBar: AppBar(
        backgroundColor: AppColors.cardLight,
        elevation: 0,
        scrolledUnderElevation: 0,
        leading: IconButton(
          icon: const Icon(Icons.arrow_back, color: AppColors.primary),
          onPressed: () => context.pop(),
        ),
        title: Text(
          l10n.artisanSelectionTitle,
          style: AppTypography.h3.copyWith(
            color: AppColors.primary,
            fontWeight: FontWeight.w600,
          ),
        ),
        centerTitle: false,
        titleSpacing: 0,
      ),
      body: async.when(
        loading: () => const Center(child: CircularProgressIndicator()),
        error: (err, _) => _ErrorView(

class _AnimatedDots extends StatefulWidget {
  const _AnimatedDots();

  @override
  State<_AnimatedDots> createState() => _AnimatedDotsState();
}

class _AnimatedDotsState extends State<_AnimatedDots>
    with SingleTickerProviderStateMixin {
  late final AnimationController _controller;

  @override
  void initState() {
    super.initState();
    _controller = AnimationController(
      vsync: this,
      duration: const Duration(milliseconds: 1400),
    )..repeat();
  }

  @override
  void dispose() {
    _controller.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return AnimatedBuilder(
      animation: _controller,
      builder: (context, _) {
        return Row(
          mainAxisSize: MainAxisSize.min,
          children: List.generate(3, (i) {
            final delay = i * 0.2;
            final t = (_controller.value - delay) % 1.0;
            final opacity = 0.3 + 0.7 * (1 - (2 * t - 1).abs());

            return Padding(
              padding: EdgeInsets.only(right: i < 2 ? 3 : 0),
              child: Opacity(
                opacity: opacity.clamp(0.3, 1.0),
                child: Container(
                  width: 4,
                  height: 4,
                  decoration: const BoxDecoration(
                    shape: BoxShape.circle,
                    color: AppColors.secondary,
                  ),
                ),
              ),
            );
          }),
        );
      },
    );
  }
}

    final selectedId = _selectedCategoryId;
    if (selectedId == null) return;

    final authState = ref.read(authStateProvider);

    if (authState is! AuthAuthenticated) {
      final result = await showLoginPromptSheet(context);

      if (!mounted) return;

      switch (result) {
        case LoginPromptResult.continueBrowsing:
          return;
        case LoginPromptResult.login:
          // Navigate to the protected route â€” the router will redirect
          // to /login?redirect=/create-request?category=... automatically
          context.go('/create-request?category=$selectedId');
          return;
        case LoginPromptResult.register:
          context.go('/create-request?category=$selectedId');
          return;

    if (!mounted) return;

    final response = await api.get('/services');

    final list = (json['services'] as List?) ?? [];

    return list
        .whereType<Map<String, dynamic>>()
        .map((entry) {
          final category = entry['category'] as Map<String, dynamic>?;
          return ServiceRef(
            id: entry['_id']?.toString() ??

            categorySlug: category?['slug']?.toString() ?? '',

          );
        })
        .where((s) => s.id.isNotEmpty)
        .toList();

    rethrow;

    throw ApiException(message: 'Failed to load services: $e');

      return _initials();

    final asyncBytes = ref.watch(artisanPhotoProvider(artisan.userId));

    return asyncBytes.when(
      loading: () => _initials(),
      error: (_, _) => _initials(),
      data: (bytes) {
        if (bytes == null || bytes.isEmpty) return _initials();
        return Image.memory(
          bytes,
          fit: BoxFit.cover,
          gaplessPlayback: true,
        );
      },
    );
  }

    final initials = _initialString();
    return Container(
      color: AppColors.primary.withValues(alpha: 0.08),

        initials,
        style: AppTypography.bodyLarge.copyWith(
          color: AppColors.primary,
          fontWeight: FontWeight.w700,
          fontSize: size * 0.34,
        ),
      ),
    );
  }

  String _initialString() {
    final f = artisan.firstName.isNotEmpty
        ? artisan.firstName[0]
        : (artisan.name.isNotEmpty ? artisan.name[0] : '?');

class DatePickerRow extends StatelessWidget {
  const DatePickerRow({
    super.key,
    required this.selectedDate,
    required this.onSelected,
    this.dayCount = 14,
  });

  final DateTime? selectedDate;
  final ValueChanged<DateTime> onSelected;
  final int dayCount;

  @override
  Widget build(BuildContext context) {
    final l10n = AppLocalizations.of(context);
    final today = DateTime.now();
    final startOfToday = DateTime(today.year, today.month, today.day);

    final dates = List<DateTime>.generate(
      dayCount,
      (i) => startOfToday.add(Duration(days: i)),
    );

    return Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        // Header row: label + month
        Row(
          mainAxisAlignment: MainAxisAlignment.spaceBetween,
          children: [
            Text(
              l10n.createRequestStep3SelectDate,
              style: AppTypography.bodyMedium.copyWith(
                color: AppColors.primary,
                fontWeight: FontWeight.w500,
              ),
            ),
            Text(
              _monthLabel(l10n, dates.first),
              style: AppTypography.bodySmall.copyWith(
                color: AppColors.secondary,
                fontWeight: FontWeight.w600,
              ),
            ),
          ],
        ),
        const SizedBox(height: AppSpacing.sm),

        // Horizontal scrollable pill row
        SizedBox(
          height: 56,
          child: ListView.separated(
            scrollDirection: Axis.horizontal,
            itemCount: dates.length,
            separatorBuilder: (_, _) => const SizedBox(width: AppSpacing.sm),
            itemBuilder: (context, index) {
              final date = dates[index];
              final isSelected =
                  selectedDate != null &&
                  selectedDate!.year == date.year &&
                  selectedDate!.month == date.month &&
                  selectedDate!.day == date.day;

              return _DatePill(
                date: date,
                isSelected: isSelected,
                onTap: () => onSelected(date),
                locale: l10n.localeName,
              );
            },
          ),
        ),

        // Confirmation line
        if (selectedDate != null) ...[
          const SizedBox(height: AppSpacing.lg),
          Text(
            l10n.createRequestStep3SelectedDate(
              _formattedSelected(l10n, selectedDate!),
            ),
            style: AppTypography.bodyMedium.copyWith(
              color: AppColors.primary,
              fontWeight: FontWeight.w500,
            ),
          ),
        ],
      ],
    );
  }

  String _monthLabel(AppLocalizations l10n, DateTime date) {
    return DateFormat.yMMMM(l10n.localeName).format(date);
  }

  String _formattedSelected(AppLocalizations l10n, DateTime date) {
    return DateFormat.yMMMMEEEEd(l10n.localeName).format(date);
  }
}

class _MapGridPainter extends CustomPainter {
  @override
  void paint(Canvas canvas, Size size) {
    final paint = Paint()
      ..color = AppColors.borderLight.withValues(alpha: 0.6)
      ..strokeWidth = 1;

    const step = 40.0;

    // Vertical lines
    for (double x = 0; x < size.width; x += step) {
      canvas.drawLine(
        Offset(x, 0),
        Offset(x, size.height),
        paint,
      );
    }

    // Horizontal lines
    for (double y = 0; y < size.height; y += step) {
      canvas.drawLine(
        Offset(0, y),
        Offset(size.width, y),
        paint,
      );
    }
  }

  @override
  bool shouldRepaint(covariant CustomPainter oldDelegate) => false;
}
